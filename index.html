// 使用 Firestore 的 onSnapshot 進行即時同步監聽 async function openPipEditor(pipId) { const { doc, onSnapshot, getFirestore } = await import("https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"); // 建立即時監聽器 onSnapshot(doc(getFirestore(), "pips", pipId), (docSnap) => { if (docSnap.exists()) { window.currentPip = docSnap.data(); // 自動即時更新主管端畫面 refreshPipUI(); switchWeekTab(window.activeWeek || 1); } }); }
