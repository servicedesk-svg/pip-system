function doPost(e) { try { var data = JSON.parse(e.postData.contents); var to = data.to; // 收件人 (員工 Email) var cc = data.cc || ""; // 副本 (主管 Email 群組) var subject = data.subject; // 信件標題 var htmlBody = data.htmlBody; // 信件內容 if (to) { // 執行發送郵件 MailApp.sendEmail({ to: to, cc: cc, subject: subject, htmlBody: htmlBody }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Email sent successfully" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // 支援 OPTIONS 預檢請求，避免跨網域被阻擋 function doOption(e) { return ContentService.createTextOutput("") .setMimeType(ContentService.MimeType.TEXT); }
